#!/usr/bin/env bash
set -euo pipefail

SSH_OPTS="-o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null -o ConnectTimeout=10"

BACKENDS=(WekaNode01 WekaNode02 WekaNode03 WekaNode04 WekaNode05)
BACKEND_IPS=(10.200.100.11 10.200.100.12 10.200.100.13 10.200.100.14 10.200.100.15)

CLIENTS=(Client01 Client02)
CLIENT_IPS=(10.200.100.21 10.200.100.22)

CLUSTER_NAME="${CLUSTER_NAME:-WekaDSXAirDemo}"
FS_GROUP="${FS_GROUP:-default}"
FS_NAME="${FS_NAME:-default}"
FS_SIZE="${FS_SIZE:-80GiB}"
MOUNT_POINT="${MOUNT_POINT:-/mnt/weka}"
BACKEND_IP="${BACKEND_IP:-10.200.100.11}"

echo "===== Configure backend data network ====="
for i in "${!BACKENDS[@]}"; do
  h="${BACKENDS[$i]}"
  ip="${BACKEND_IPS[$i]}"
  echo "===== $h / $ip ====="
  ssh $SSH_OPTS ubuntu@$h "sudo bash -s" <<EOS
set -e
cat > /etc/netplan/70-weka-data.yaml <<NETPLAN
network:
  version: 2
  renderer: networkd
  ethernets:
    eth1:
      mtu: 9000
      dhcp4: false
      dhcp6: false
      addresses:
        - ${ip}/24
NETPLAN
chmod 600 /etc/netplan/70-weka-data.yaml
chmod 600 /etc/netplan/40-air.yaml 2>/dev/null || true
netplan generate
netplan apply
ip -br a | egrep "eth0|eth1"
EOS
done

echo "===== Configure client data network ====="
for i in "${!CLIENTS[@]}"; do
  c="${CLIENTS[$i]}"
  ip="${CLIENT_IPS[$i]}"
  echo "===== $c / $ip ====="
  ssh $SSH_OPTS ubuntu@$c "sudo bash -s" <<EOS
set -e
cat > /etc/netplan/70-weka-client-data.yaml <<NETPLAN
network:
  version: 2
  renderer: networkd
  ethernets:
    eth1:
      mtu: 9000
      dhcp4: false
      dhcp6: false
      addresses:
        - ${ip}/24
NETPLAN
chmod 600 /etc/netplan/70-weka-client-data.yaml
chmod 600 /etc/netplan/40-air.yaml 2>/dev/null || true
netplan generate
netplan apply
ip -br a | egrep "eth0|eth1"
EOS
done

echo "===== Validate data network from WekaNode01 ====="
for ip in "${BACKEND_IPS[@]}" "${CLIENT_IPS[@]}"; do
  ping -c 2 "$ip"
done

echo "===== Prepare backend packages, swap, and drives ====="
for h in "${BACKENDS[@]}"; do
  echo "===== $h ====="
  ssh $SSH_OPTS ubuntu@$h "sudo bash -s" <<'EOS'
set -e
systemctl stop unattended-upgrades apt-daily.service apt-daily-upgrade.service 2>/dev/null || true
systemctl stop apt-daily.timer apt-daily-upgrade.timer 2>/dev/null || true
apt-get update -y
apt-get install -y curl jq fio lsof nvme-cli

# swap disk = nvme1n1, WEKA data disk = nvme0n1
swapoff /dev/nvme1n1 2>/dev/null || true
wipefs -a /dev/nvme1n1 || true
mkswap /dev/nvme1n1
swapon /dev/nvme1n1

wipefs -a /dev/nvme0n1 || true
dd if=/dev/zero of=/dev/nvme0n1 bs=1M count=100 conv=fsync status=none
blockdev --rereadpt /dev/nvme0n1 2>/dev/null || true

free -h
nvme list || true
EOS
done

echo "===== Create backend containers in UDP mode ====="
for i in "${!BACKENDS[@]}"; do
  h="${BACKENDS[$i]}"
  ip="${BACKEND_IPS[$i]}"

  if [[ "$i" == "0" ]]; then
    cores=3
    mem="4GiB"
  elif [[ "$i" == "1" ]]; then
    cores=2
    mem="2GiB"
  else
    cores=2
    mem="3GiB"
  fi

  echo "===== $h $ip cores=$cores mem=$mem ====="
  ssh $SSH_OPTS ubuntu@$h "sudo bash -s" <<EOS
set -e
weka local stop -f 2>/dev/null || true
weka local rm --all --force 2>/dev/null || true

weka local setup container \
  --name default \
  --base-port 14000 \
  --cores ${cores} \
  --memory ${mem} \
  --management-ips ${ip} \
  --join-ips ${BACKEND_IP}

weka local start default
weka local ps
EOS
done

echo "===== Create cluster ====="
weka cluster create "${CLUSTER_NAME}" "${BACKEND_IPS[@]}"
sleep 10
weka status

echo "===== Add backend drives using container IDs ====="
for id in 0 1 2 3 4; do
  echo "===== adding /dev/nvme0n1 to container $id ====="
  weka cluster drive add "$id" /dev/nvme0n1
done

weka cluster drive

echo "===== Start IO ====="
weka cluster start-io
sleep 10
weka status

echo "===== Create filesystem ====="
weka fs group create "$FS_GROUP" || true
if ! weka fs | awk '{print $1}' | grep -qx "$FS_NAME"; then
  weka fs create "$FS_NAME" "$FS_GROUP" "$FS_SIZE"
fi
weka fs

echo "===== Mount clients using UDP ====="
for i in "${!CLIENTS[@]}"; do
  c="${CLIENTS[$i]}"
  ip="${CLIENT_IPS[$i]}"
  echo "===== $c / $ip ====="
  ssh $SSH_OPTS ubuntu@$c "sudo bash -s" <<EOS
set -e
apt-get update -y
apt-get install -y curl jq fio lsof

umount ${MOUNT_POINT} 2>/dev/null || true
weka local stop client 2>/dev/null || true
weka local rm client 2>/dev/null || true
weka local stop client2 2>/dev/null || true
weka local rm client2 2>/dev/null || true
sleep 5

mkdir -p ${MOUNT_POINT}

ping -c 2 ${BACKEND_IP}
nc -vz ${BACKEND_IP} 14000

mount -t wekafs \
  -o net=udp,num_cores=0,mgmt_ip=${ip} \
  ${BACKEND_IP}/${FS_NAME} \
  ${MOUNT_POINT}

chmod 777 ${MOUNT_POINT}
df -h ${MOUNT_POINT}
mount | grep wekafs
weka local ps
EOS
done

echo "===== fio smoke test ====="
for c in "${CLIENTS[@]}"; do
  echo "===== fio from $c ====="
  ssh $SSH_OPTS ubuntu@$c "
    mkdir -p ${MOUNT_POINT}/fio-test &&
    fio --name=weka-client-write-${c} \
      --directory=${MOUNT_POINT}/fio-test \
      --rw=write \
      --bs=1M \
      --size=2G \
      --numjobs=2 \
      --iodepth=8 \
      --ioengine=libaio \
      --direct=1 \
      --runtime=30 \
      --time_based \
      --group_reporting
  "
done

echo "===== Final status ====="
weka status
weka cluster drive
weka fs
